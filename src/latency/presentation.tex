\section{Trading}
\begin{frame}{Electronic Trading}
	\begin{itemize}
		\item Electronic trading involves interacting with a \textbf{matching engine}.
		\item A \textbf{limit order} represents an intention to trade some \textbf{quantity} in some \textbf{direction} at some \textbf{limit price} or better
		\item We can send three kinds of messages to the matching engine:
		\begin{itemize}
			\item \textbf{Insert} an order with some \textbf{lifetime} (e.g. GTC, GFD, FAK, FOK)
			\item \textbf{Cancel} or \textbf{modify} existing order(s)
		\end{itemize}
		\item The matching engine maintains an \href{https://www.binance.com/en-AU/trade/BTC_USDT}{\textbf{order book}} and produces \textbf{matches} when two orders are willing to trade with each other
	\end{itemize}
	A market is a distributed computer for producing trades at mutually agreeable prices.
\end{frame}

\begin{frame}{Trading Strategies}
	\begin{block}{Arbitrage}
		\begin{itemize}
			\item \textbf{Arbitrage} refers to making offsetting trades in different order books, like buying a stock on one exchange and selling it on another
			\item This often involves a \textbf{race condition}, with the risk of getting `\textbf{legged}'
		\end{itemize}
	\end{block}
	\begin{block}{Risk}
		\begin{itemize}
			\item \textbf{Statistical Arbitrage}: Positions might not offset perfectly
			\item \textbf{Speculation}: We might make a trade with the expectation of offsetting it at some later time
			\item Traders have varying \textbf{risk profiles}, which causes disagreement about which trades are favourable
			\item \textbf{Liquidity}: The more favourable trades we have access to, the more we can make, and the faster we can trade out of a risky position
		\end{itemize}
	\end{block}

\end{frame}

\begin{frame}{Toy model}
	Expected quadratic utility of trading $q$ units with a forecast horizon $h$, forecast $\mu$, volatility $\sigma$, and risk aversion $\lambda$ is roughly given by
	$$q \mu-\lambda q^2 \sigma^2(h+t),$$
	where $t$ is the time taken to get out of the position at a fair price.
\end{frame}

\begin{frame}{Low-Latency Trading}
	\begin{itemize}
		\item Access to \textbf{better liquidity} and understanding of how to make use of it represents a concrete, measurable edge.
		\item Almost any profitable trading strategy can be improved by being faster.
		\item Latency helps the most with the \textbf{most competitive opportunities}
		\item \textbf{Capacity constraints}: Very large trading businesses may become less latency-focused as risk budgets expand
		\item Market structure understanding remains relevant for understanding order flow
	\end{itemize}
\end{frame}

\section{Time}

\begin{frame}{Clocks}
	\begin{itemize}
		\item Quartz powered by a DC signal vibrates due to the piezoelectric effect at a reliable \textbf{reference frequency}. \\
			\begin{center} \includegraphics[scale=0.04]{assets/clockwork.jpg} \includegraphics[scale=0.2]{assets/crystal.jpg} \end{center}
		\item A \textbf{voltage-controlled oscillator} has a frequency dependent on the input voltage.
		\item A \textbf{phase detector} can produce a signal whose strength depends on how fast or slow the VCO is running compared to the reference.
		\item By keeping the VCO in a \textbf{phase-locked loop} with the reference, we can achieve oscillation at some multiple of the reference frequency. This can be chained a number of times.
	\end{itemize}
\end{frame}

\begin{frame}{Clock Drift}
	Heat, manufacturing differences, crystal aging, voltage fluctuations, etc., can cause two clocks $T_1(t)$ and $T_2(t)$ to experience \textbf{clock drift}, whereby $\theta_{1\to 2} = T_2(t)-T_1(t)$ changes over time.

	\begin{center}
		\includegraphics[scale=0.4]{assets/clock_drift.png}
	\end{center}

	It is possible to observe negative apparent latency when naively comparing timestamps from poorly synced clocks.

	%Note that this is not trivial to distinguish from slow changes in service quality over time. But generally those are a bit more discrete
\end{frame}

\begin{frame}{Synchronisation}
	For synchronisation within a network we will typically have a time server with a \textbf{master clock}. This could be set just once or synced with some external source like GPS.

	\begin{center}
		\begin{tikzpicture}[
		    node distance=2cm,

		]
		    \node (switch) at (0,0) {switch};
		    \node (master) at (3,0) {time server};

		    \foreach \y in {1,0,-1} {
			\node (client\y) at (-3,\y) {client};
			\draw (client\y) -- (switch);
		    }

		    \draw (switch) -- (master);
		\end{tikzpicture}
	\end{center}

	\begin{block}{Clock Drift Estimation}
		Each piece of hardware can be equipped with its own clock $T_{*}(t)$.

		We can \textbf{send messages} over the network, \textbf{record timestamps} at each device clock, and \textbf{adjust for latency}.

	\end{block}
	This allows us to adjust the client clock speed (\textbf{slewing}) such that $\theta_{c\to m}$ approaches zero.
\end{frame}

\begin{frame}{Network Time Protocol}
	Using the \textbf{network time protocol} involves the client making requests to the time server that are labeled with four receive/transmit timestamps $T_c(t_1), T_m(t_2), T_m(t_3), T_c(t_4)$.

	Suppose we have symmetric latency $T_m(t_2)-T_m(t_1) = T_m(t_4)-T_m(t_3)$.
	It follows that $T_m(t_2)-(T_c(t_1)+\theta_{c\to m}) = (T_c(t_4)+\theta_{c\to m})-T_m(t_3)$. Then
	\begin{align*}
		\theta_{c\to m} &= \frac{1}{2}\left(\underbrace{(T_m(t_2)-T_c(t_1))}_\text{Transmit} - \underbrace{(T_c(t_4)-T_m(t_3))}_\text{Receive}\right) \\
		&= \underbrace{\frac{1}{2}(T_m(t_3)+T_m(t_2))}_{\text{Midpoint for }m} - \underbrace{\frac{1}{2}(T_c(t_4)+T_c(t_1))}_{\text{Midpoint for }c} \\
	\end{align*}

	In the case where wire-to-wire latency is zero, this is also known as Einstein synchronisation.
\end{frame}

\begin{frame}{Precision Time Protocol}
	Because the latency added by a switch is nondeterministic, we can achieve greater precision by controlling for its contribution.

	Suppose we have $T_c(t_1), T_s(t_2), T_s(t_3), T_m(t_4), T_m(t_5), T_s(t_6), T_s(t_7), T_c(t_8)$.

	Then we have
	$$\theta_{c\to s} = \frac{1}{2}\left((T_s(t_2)-T_c(t_1)) - (T_c(t_8)-T_s(t_7))\right),$$
	$$\theta_{s\to m} = \frac{1}{2}\left((T_m(t_4)-T_s(t_3)) - (T_s(t_6)-T_m(t_5))\right),$$
	$$\theta_{c\to m} = \theta_{c\to s} + \theta_{s\to m}.$$

	We can either record a cumulative adjustment in the timing packets (transparent clock), or sync the switch to the master clock and then the client to the switch (boundary clock).
\end{frame}

\begin{frame}{Network Timestamping}

	\begin{itemize}
		\item Timestamping of messages may be done at multiple points in the system (e.g. \texttt{timestamp\_match} and \texttt{timestamp\_publication}).

	\end{itemize}

\begin{frame}{Fragmentation}
	\begin{block}{}
		``The virtual channels on the X.25 for ordering were only 64kbps – very slow. I figured we could hack those similarly \ldots and start filling out an order before we had one. This would save us milliseconds. You could turn the packet into something benign or invalid if you didn’t have an order.'' Matt Hurd

		\item Exchanges typically monitor packet corruption rates for signs of abuse
	\end{itemize}

\begin{frame}{Lottery Game}
	\begin{block}{}
		\begin{itemize}
			\item An online lottery website is offering a \$1m jackpot.
			\item Unfortunately, because they vibecoded the website, nobody is able to buy any tickets. Even worse, $n$ users have each been given one free ticket.
			\item Each number $i$ has some probability $p_i$ of winning, in which case the jackpot will be split equally amongst all players who chose $i$. If nobody wins, the jackpot will not pay out.
		\end{itemize}

		How should the players choose their numbers?
	\end{block}

\end{frame}

\begin{frame}{Lottery Game Solution}
	If a player chooses each number $i$ with probability $\pi_i$, and other players with probability $\pi_i'$, the player's expected payoff (in millions of dollars) is

	\begin{align*}
			& \pi_i \cdot p_i \cdot \left(\sum_{k=0}^{n-1}\frac{1}{1+k}\binom{n-1}{k} \pi_i'^k (1-\pi_i')^{n-1-k}\right) \\
		=	& \pi_i \cdot p_i \cdot \left(\sum_{k=0}^{n-1} \left(\int_0^1 x^k dx\right)\binom{n-1}{k} (\pi_i')^k(1-\pi_i')^{n-1-k} \right) \\
		=	& \pi_i \cdot p_i \cdot \int_0^1 \left(\sum_{k=0}^{n-1} \binom{n-1}{k} (\pi_i'\cdot x)^k (1-\pi_i')^{n-1-k} \right)dx \\
		=	& \pi_i \cdot p_i \cdot \int_0^1 (1 - \pi_i' + \pi_i'\cdot x)^{n-1} dx = \pi_i \cdot p_i \cdot \frac{1 - (1-\pi_i')^n}{n\pi_i'}.
	\end{align*}

	For a symmetric equilibrium, we require that $p_i \cdot \frac{1 - (1-\pi_i')^n}{n\pi_i'}$ is the same for all $i$. If $(1-\pi_i')^n$ is negligible this reduces to $\pi_i'\propto p_i$.
\end{frame}

\begin{frame}{CPUs}
	Logically, a Central Processing Unit runs the following loop:
	\begin{enumerate}
		\item \textbf{Fetch} the program instruction indicated by the \textbf{program counter}
		\item \textbf{Decode} the instruction into primitive operations %micro-operations
		\item \textbf{Execute} the instruction, which may involve:
		\begin{itemize}
			\item Computing some value
			\item Interacting with external \textbf{memory} (or cache)
			\item Altering the state of the CPU's \textbf{registers}
		\end{itemize}
	\end{enumerate}

	The \textbf{Instruction Set Architecture} (ISA) defines how \href{https://godbolt.org/z/e3q4rox9e}{\textbf{machine instructions}} are interpreted.

	You can check what CPU you have with \texttt{lscpu}.

\end{frame}

\begin{frame}{Overclocking}
	\begin{itemize}
		\item The Fetch-Decode-Execute loop requires some number of clock cycles for each instruction.

		\item Naively, the amount of time a CPU takes to achieve a task will be the total number of clock cycles required times the clock period.

		\item For maximum performance, CPUs can be \textbf{overclocked} relative to the speed they are designed for.

		\item This generates heat in excess of what the CPU is designed to handle, so an appropriate \textbf{cooling system} is required.%not great if the rack is packed with stuff

		\item It also increases the \textbf{power draw} of the system.
	\end{itemize}

	Clock frequency can be monitored using \texttt{turbostat}, e.g. \texttt{sudo turbostat -{}-quiet -{}-show Core,CPU,Busy\%,Bzy\_MHz,Avg\_MHz,IPC}.

	OS-level control of clock frequency can be configured using \texttt{cpupower}.
%https://www.blackcoretech.com/
\end{frame}

\begin{frame}{Performance Optimisation}
	\begin{itemize}
		\item Make sure to \textbf{do as little work as possible}
		\item Schedule tasks in a way that lets you \textbf{have many going at once}
		\item Arrange memory so you can get away with \textbf{communicating as little as possible}
		\item Use profiling tools to \textbf{constantly measure and compare}
	\end{itemize}

\end{frame}

\begin{frame}{Out-of-Order Execution}
	\begin{block}{Parallelism}
		In practice, the Fetch-Decode-Execute loop is parallelised in modern CPUs.

		\begin{enumerate}
			\item \textbf{Issue} the instruction to a relevant \textbf{reservation station}
			\item Wait for the operands to be ready before \textbf{executing} the operation
			\item \textbf{Write} the result to the output register and \textbf{broadcast} to other reservation stations
		\end{enumerate}
	\end{block}

	\begin{itemize}
		\item Taking advantage of parallelism involves \textbf{dependency minimisation}. Modern compilers are quite good at this.

		\item CPUs are not always transparent about their inner workings
		\item Tools like \href{https://godbolt.org/z/8P6jKv81c}{\texttt{llvm-mca --timeline}} can give some very rough hints
	\end{itemize}
\end{frame}

\begin{frame}{x86}
	\begin{itemize}
		\item An \textbf{Instruction Set Architecture} defines the available \textbf{machine instructions} and their behaviour.

		\item The ISA also exposes a fixed set of \href{https://en.wikibooks.org/wiki/X86_Assembly/X86_Architecture}{\textbf{logical registers}}.
	\end{itemize}

	\begin{block}{x86}
		\begin{itemize}
			\item ISA developed by Intel in 1978, largely backwards-compatible
			\item Variable-length instructions%denser code packing
			\item Instructions often implemented using microoperations ($\mu$ops)
			\item Support for Single Instruction Multiple Data (SIMD)
		\end{itemize}
	\end{block}

\begin{frame}{Storing Local Variables}
	\begin{block}{Register Spilling}
		\begin{itemize}
			\item Having many variables in scope can lead to \textbf{register pressure}
			\item To cope with this, variables may be \href{https://godbolt.org/z/6Mcdac7qY}{\textbf{spilled}} onto \textbf{stack memory}
		\end{itemize}
	\end{block}

	\begin{block}{Stack Memory}
		\begin{itemize}
			\item A fixed amount of stack memory is allocated when the program starts
			\item The \textbf{stack pointer} (\texttt{rsp} on x86) represents the region of data used in a function scope
			\item x86 supports \texttt{push} and \texttt{pop} operations
			\item At the start of a function call:
			\begin{itemize}
				\item Registers that will be used are pushed to stack
				\item Register \texttt{rsp} is decremented by the space required for storing local variables
			\end{itemize}
			\item Moving \texttt{rsp} outside the allocated range of stack memory represents a \href{https://godbolt.org/z/85TTjWKq7}{\textbf{stack overflow}}
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}{Heap Memory}
	\begin{itemize}
		\item Sometimes the amount of memory we need is only known at runtime
		\item We can request \textbf{virtual memory} from the OS using syscalls like \texttt{brk} and \texttt{mmap}
		\item Since syscalls are expensive, we can use a software-level allocator like \texttt{malloc} that fetches a large amount of memory in advance
		\item Allocation is still expensive and should be avoided in the hotpath
		\item \textbf{Heap fragmentation} and \textbf{pressure} can slow down allocation even further
	\end{itemize}

\end{frame}

\begin{frame}{Caching}
	\begin{itemize}
		\item Register access is much faster than RAM access
		\item \textbf{Caches} provide intermediate \textbf{access speeds} and \textbf{memory capacity} by \textbf{cache lines}
		\item Caches use \href{https://uops.info/cache.html}{\textbf{eviction rules}} (e.g. Least Recently Used) to determine which data to keep in memory %makes sense for stack
		\item Caches are organised \textbf{hierarchically}: L1, L2, L3 (+ RAM, swap space)
		\item \textbf{Cache misses} are more expensive
	\end{itemize}

	Cache details are shown by \texttt{lscpu}.
\end{frame}

\begin{frame}{Branch Prediction}
	\begin{itemize}
		\item Often the program counter needs to \textbf{jump} depending on some condition
		\item In order to allow pipelining before this condition is resolved, CPUs use \textbf{speculative execution} based on a \textbf{branch predictor}
	\end{itemize}

	\begin{block}{Branch Prediction Rules}
		\begin{itemize}
			\item Static: always jump, jump if backwards
			\item Local history: last outcome, saturating counter
			\item Global history
			\item Random
			\item Learned: perceptron, hash buckets, etc.
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}{Branch Misses}
	\begin{itemize}
		\item We can count branch misses with \texttt{perf stat -e branches,branch-misses}
		\item Figuring out which branches were mispredicted can be nontrivial and quite CPU-specific
		\item Sometimes it is cheaper to compute both branches and use \texttt{cmov}
		\item We can hint to the compiler with keywords like \texttt{[[likely]]}/\texttt{[[unlikely]]}. Instruction-level branch hints are not normally supported.
	\end{itemize}

\begin{frame}{Multi-Core Processing}
	\begin{itemize}
		\item Modern processors come equipped with multiple \textbf{cores}
		\item Each core may have multiple CPUs, visible with \texttt{lscpu -e=CPU,CORE}
		\item Each CPU can execute instructions independently, e.g. for multithreaded code
		\item Threads may be \textbf{pinned} to a core, or have an \textbf{affinity bitmask} (\texttt{taskset}, \texttt{cpuset}, \texttt{tuna})
		\item \textbf{Interrupt Request} (IRQ) affinity (particularly NIC IRQs) and steering may assist performance
		\item Configuration parameters like \texttt{isolcpus} and \texttt{nohz\_full} allow a core to avoid running certain tasks
	\end{itemize}

	\begin{block}{Interprocess Communication (IPC)}
		Different processes often need to communicate information to one another.

		Libraries like \texttt{boost::interprocess} rely on the \texttt{mmap} syscall to allocate \textbf{shared memory}.
	\end{block}
\end{frame}

\begin{frame}{Cache Coherence}
	\begin{itemize}
		\item Each \textbf{cache line} is \textbf{single-writer} to ensure \textbf{coherence}
		\item Transferring ownership is costly
		\item \textbf{False Sharing}: Threads modifying data in the same cache line causes ownership to `bounce' between cores
		\item Cache writes trigger \textbf{cache invalidation} for other cores reading the cache line
		\item It may be possible to configure the NIC to share a cache with the processors
	\end{itemize}
	
	\begin{block}{}
		``There are 2 hard problems in computer science: cache invalidation, naming things, and off-by-1 errors.'' Leon Bambrick
	\end{block}
\end{frame}

\begin{frame}[fragile]{Race Conditions}
	Even with single-writer, \textbf{data races} are still possible if operations are \textbf{non-atomic}. Using a \textbf{mutex} also prevents this.

\begin{lstlisting}[ basicstyle=\ttfamily\bfseries\scriptsize ]
mov    eax, DWORD PTR [x]    ; load x
add    eax, 1                ; x + 1
mov    DWORD PTR [x], eax    ; store x
\end{lstlisting}

	The atomic version would be:
\begin{lstlisting}[ basicstyle=\ttfamily\bfseries\scriptsize ]
lock add DWORD PTR [x], 1
\end{lstlisting}

	A \textbf{producer-consumer} queue can help prevent data races and balance load across threads.
\end{frame}

\begin{frame}{Multitasking}
	OS multitasking is achieved using:
	\begin{itemize}
		\item A \textbf{priority mechanism}, set using \texttt{chrt}\footnote{e.g. \texttt{sched\_other}/\texttt{sched\_normal}, \texttt{sched\_batch}, \texttt{sched\_fifo}, \texttt{sched\_rr}, \texttt{sched\_deadline}}
		\item A \textbf{task-switching mechanism}, e.g. time slicing, blocks/yields, interrupts
	\end{itemize}

	\begin{block}{Types of Multitasking}
		\begin{itemize}
			\item Multitasking governs both \textbf{multiprocessing} or \textbf{multithreading} (e.g. \texttt{std::thread}).
			\item Alternatively, userspace multitasking can be achieved using \textbf{coroutines} (e.g. \texttt{boost::fibers::fiber}).
		\end{itemize}
	\end{block}

	\begin{itemize}
		\item Task switching is relatively expensive
		\item Interrupt-based communication should generally be avoided, e.g. \texttt{IRQBALANCE\_BANNED\_CPUS}%interrupt storms
	\end{itemize}
\end{frame}

\begin{frame}{Userspace Networking}
	\begin{itemize}
		\item Interrupt-based networking introduces \textbf{task-switching overhead}
		\item A standard NIC setup will \textbf{not interrupt immediately} upon receipt of a message
		\item We can \texttt{mmap} the NIC's DMA segment to use \textbf{shared memory} instead
		\item \textbf{Continuous polling} of the RX queue allows for rapid response%cache warming
	\end{itemize}

	\begin{block}{Kernel Bypass Frameworks}
		\begin{itemize}
			\item We need to know the layout of DMA to communicate with the NIC
			\item TCPDirect, DPDK (+F-Stack/mTCP) provide specialised APIs
			\item OpenOnload and Nvidia Messaging Accelerator make use of \texttt{LD\_PRELOAD} to monkeypatch\footnote{Specifically, changing the dynamic linking} \texttt{libc}'s socket calls
		\end{itemize}
	\end{block}
\end{frame}

		\item Market impact
		\item Risk management and controls
		\item Human oversight
		\item Financing
